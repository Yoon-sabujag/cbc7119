async function handleDownload(category: string, type: 'inspect' | 'public') {
  const key = `${category}-${type}`
  setBusy(key)
  try {
    const res  = await fetch(`/api/checkpoints?category=${encodeURIComponent(category)}`, {
      headers: { Authorization: `Bearer ${token}` },
    })
    const json = await res.json() as { success: boolean; data?: CheckPoint[] }
    if (!json.success || !json.data?.length) {
      toast.error('체크포인트가 없습니다')
      return
    }
    const catLabel = CATEGORIES.find(c => c.value === category)?.label ?? category
    await generatePdf(json.data, type, baseUrl, catLabel)
    toast.success('PDF 다운로드 완료')
  } catch (e) {
    toast.error('PDF 생성 오류')
    console.error(e)
  } finally {
    setBusy(null)
  }
}

function dlBtnStyle(loading: boolean, isPublic: boolean): React.CSSProperties {
  return {
    padding: '9px 16px',
    borderRadius: 9,
    border: 'none',
    background: loading
      ? 'var(--bd2)'                                    // -> bg-border-strong (loading)
      : isPublic
        ? 'linear‑gradient(135deg,#16a34a,#22c55e)'  // 폐기 (OQ #1 LOCKED) → solid bg-status-safe-bar
        : 'linear-gradient(135deg,#1d4ed8,#0ea5e9)',    // -> inspect 유지 (W4 만 예외 anchor)
    color: loading ? 'var(--t3)' : '#fff',              // -> text-text-tertiary (loading) / text-text-on-accent (활성)
    fontSize: 12,                                        // -> text-caption
    fontWeight: 700,                                     // -> font-bold
    cursor: loading ? 'default' : 'pointer',
    display: 'flex',
    alignItems: 'center',
    gap: 6,
  }
}
